<script>
(function(){
  var SHEET_ID="1O2u1iakOHHv-jG8EUOOJ2iVhAsjibxNxxSQ-LP6yDDI", SHEET_NAME="EXPANDED DATA";
  var MON=["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  var data=[], flt={month:"all",delivery:"all",status:"all",company:"all"}, charts={};
  var $=function(id){return document.getElementById(id)};
  var fmt=function(n){return Math.round(n).toLocaleString("en-IN")};
  function num(v){var n=parseFloat(String(v==null?"":v).replace(/,/g,""));return isNaN(n)?0:n}
  function pd(s){var m=/^(\d{1,2})\/(\d{1,2})\/(\d{4})/.exec(String(s||"").trim());return m?{y:+m[3],m:+m[2]}:null}
  function mkey(d){return d?d.y+"-"+(d.m<10?"0":"")+d.m:""}
  function mlabel(k){var p=k.split("-");return MON[+p[1]-1]+" "+p[0]}
  function css(n){return getComputedStyle(document.documentElement).getPropertyValue(n).trim()}
  function esc(s){return String(s).replace(/[&<>"]/g,function(c){return{"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]})}
  function trunc(s,n){s=String(s);return s.length>n?s.slice(0,n-1)+"…":s}

  function parseCSV(t){
    var rows=[],row=[],f="",q=false;
    for(var i=0;i<t.length;i++){var c=t[i];
      if(q){ if(c==='"'){ if(t[i+1]==='"'){f+='"';i++}else q=false } else f+=c }
      else if(c==='"')q=true; else if(c===","){row.push(f);f=""}
      else if(c==="\n"||c==="\r"){ if(c==="\r"&&t[i+1]==="\n")i++; row.push(f);f="";rows.push(row);row=[] }
      else f+=c }
    if(f!==""||row.length){row.push(f);rows.push(row)}
    return rows;
  }

  function toRows(vals){
    var out=[];
    for(var i=1;i<vals.length;i++){
      var r=vals[i]||[]; var co=String(r[3]||"").trim(); if(!co&&!r[5]&&!r[7])continue;
      var dv=String(r[12]||"").trim().toLowerCase(), q=String(r[16]||"").trim().toLowerCase(), st=String(r[15]||"").trim().toLowerCase();
      out.push({
        po:String(r[1]||"").trim(), poDate:pd(r[2]), poDateRaw:String(r[2]||""), company:co,
        orderQty:num(r[5]), sentQty:num(r[7]), dispDate:pd(r[8]),
        pendItem:String(r[10]||"").trim(), pendQty:num(r[11]),
        delivery:dv.indexOf("on time")===0?"On Time":(dv.indexOf("delay")===0?"Delayed":""),
        pending:q.indexOf("pend")===0||(!q&&st&&st!=="done")
      });
    }
    return out;
  }

  function notice(text){
    var n=$("notice"); n.hidden=false; n.className="msg";
    n.innerHTML="<p></p>"; n.firstChild.textContent=text;
    var b=document.createElement("button");b.className="btn";b.type="button";b.textContent="Try again";b.onclick=load;n.appendChild(b);
  }

  async function load(){
    $("notice").hidden=true; $("sub").textContent="Sheet: "+SHEET_NAME+" · loading…";
    try{
      var url="https://docs.google.com/spreadsheets/d/"+SHEET_ID+"/gviz/tq?tqx=out:csv&headers=1&sheet="+encodeURIComponent(SHEET_NAME);
      var rs=await fetch(url); if(!rs.ok)throw new Error("HTTP "+rs.status);
      var tx=await rs.text(); if(tx.trim().charAt(0)==="<")throw new Error("sheet is not public");
      data=toRows(parseCSV(tx));
      $("sub").textContent="Sheet: "+SHEET_NAME+" · "+fmt(data.length)+" lines · updated "+new Date().toLocaleTimeString("en-IN",{hour:"2-digit",minute:"2-digit"});
      buildFilters(); render();
    }catch(e){
      notice("Sheet se data nahi aaya ("+(e&&e.message||"error")+"). Google Sheet me Share > General access > 'Anyone with the link' > Viewer karein, phir Try again dabayein.");
      $("sub").textContent="Sheet: "+SHEET_NAME+" · not loaded";
    }
  }

  function chipRow(id,key,items){
    var el=$(id); el.innerHTML="";
    items.forEach(function(it){
      var b=document.createElement("button"); b.type="button"; b.className="chip"; b.textContent=it.label;
      b.setAttribute("aria-pressed",String(flt[key]===it.value));
      b.onclick=function(){flt[key]=it.value; chipsRefresh(); render()};
      b._k=key; b._v=it.value; el.appendChild(b);
    });
  }
  function chipsRefresh(){
    document.querySelectorAll(".chip").forEach(function(b){b.setAttribute("aria-pressed",String(flt[b._k]===b._v))});
  }
  function buildFilters(){
    var months={},cos={};
    data.forEach(function(r){ if(r.poDate)months[mkey(r.poDate)]=1; if(r.company)cos[r.company]=(cos[r.company]||0)+r.orderQty });
    var mk=Object.keys(months).sort();
    chipRow("fMonth","month",[{label:"All",value:"all"}].concat(mk.map(function(k){return{label:mlabel(k),value:k}})));
    chipRow("fDelivery","delivery",[{label:"All",value:"all"},{label:"On Time",value:"On Time"},{label:"Delayed",value:"Delayed"}]);
    chipRow("fStatus","status",[{label:"All",value:"all"},{label:"Delivered",value:"done"},{label:"Pending",value:"pending"}]);
    var cl=Object.keys(cos).sort(function(a,b){return cos[b]-cos[a]});
    chipRow("fCompany","company",[{label:"All",value:"all"}].concat(cl.map(function(c){return{label:c,value:c}})));
  }
  $("reset").onclick=function(){flt={month:"all",delivery:"all",status:"all",company:"all"};chipsRefresh();render()};
  $("toggleF").onclick=function(){
    var box=$("filters"); box.hidden=!box.hidden;
    $("toggleF").textContent=box.hidden?"Show filters":"Hide filters";
  };
  $("refresh").onclick=load;

  function pass(r,skipMonth){
    if(!skipMonth&&flt.month!=="all"&&mkey(r.poDate)!==flt.month)return false;
    if(flt.delivery!=="all"&&r.delivery!==flt.delivery)return false;
    if(flt.status==="pending"&&!r.pending)return false;
    if(flt.status==="done"&&r.pending)return false;
    if(flt.company!=="all"&&r.company!==flt.company)return false;
    return true;
  }
  function empty(id,t){ if(charts[id]){charts[id].destroy();delete charts[id]} $(id).innerHTML='<div class="empty">'+t+'</div>' }
  function draw(id,cfg){
    if(charts[id])charts[id].destroy();
    $(id).innerHTML='<canvas></canvas>';
    cfg.options=cfg.options||{}; cfg.options.responsive=true; cfg.options.maintainAspectRatio=false;
    charts[id]=new Chart($(id).firstChild,cfg);
  }
  function kpi(l,v,n){return '<div class="kpi"><div class="l">'+l+'</div><div class="v">'+v+'</div><div class="n">'+n+"</div></div>"}

  function render(){
    var rows=data.filter(function(r){return pass(r,false)});
    var exMonth=data.filter(function(r){return pass(r,true)});
    var mut=css("--muted"),line=css("--line"),acc=css("--accent"),acc2=css("--accent-2"),good=css("--good"),bad=css("--bad"),warn=css("--warn");
    if(window.Chart){Chart.defaults.color=mut;Chart.defaults.font.family=css("--font-body");Chart.defaults.borderColor=line}

    var orderTot=0,sent=0,on=0,del=0,pendQty=0,pendLines=0;
    rows.forEach(function(r){orderTot+=r.orderQty;sent+=r.sentQty;if(r.delivery==="On Time")on++;if(r.delivery==="Delayed")del++;if(r.pending){pendQty+=r.pendQty;pendLines++}});
    var pct=on+del?Math.round(on*1000/(on+del))/10:0, pendPct=orderTot?Math.round(pendQty*1000/orderTot)/10:0;
    $("kpis").innerHTML=
      kpi("Total order qty",fmt(orderTot),"column F, "+fmt(rows.length)+" lines")+
      kpi("Dispatched qty",fmt(sent),"column H")+
      '<div class="kpi"><div class="l">On time</div><div class="v">'+pct+'%</div><div class="n">'+fmt(on)+" on time, "+fmt(del)+' delayed</div><div class="bar"><i style="width:'+pct+'%"></i></div></div>'+
      kpi("Pending qty",fmt(pendQty),fmt(pendLines)+" lines pending"+(orderTot?", "+pendPct+"% of order qty":""));
    if(!window.Chart)return;

    var byCo={}; rows.forEach(function(r){if(r.company)byCo[r.company]=(byCo[r.company]||0)+r.orderQty});
    var cl=Object.keys(byCo).sort(function(a,b){return byCo[b]-byCo[a]}).slice(0,10);
    if(!cl.length||!orderTot)empty("cvCompany","No orders for these filters.");
    else draw("cvCompany",{type:"bar",data:{labels:cl,datasets:[{label:"Order qty",data:cl.map(function(c){return byCo[c]}),backgroundColor:acc,borderRadius:4}]},
      options:{indexAxis:"y",plugins:{legend:{display:false},tooltip:{callbacks:{label:function(c){return " "+fmt(c.parsed.x)}}}},
        scales:{x:{ticks:{callback:function(v){return fmt(v)}},grid:{color:line}},y:{ticks:{callback:function(v){return trunc(this.getLabelForValue(v),24)}},grid:{display:false}}}}});

    if(!on&&!del)empty("cvTime","No delivery results for these filters.");
    else draw("cvTime",{type:"doughnut",data:{labels:["On Time","Delayed"],datasets:[{data:[on,del],backgroundColor:[good,bad],borderColor:css("--surface"),borderWidth:2}]},
      options:{cutout:"62%",plugins:{legend:{position:"bottom"},tooltip:{callbacks:{label:function(c){return " "+c.label+": "+fmt(c.parsed)+" lines ("+Math.round(c.parsed*1000/(on+del))/10+"%)"}}}}}});

    var om={},dm={},dc={},seen={};
    exMonth.forEach(function(r){
      if(r.poDate){var k=mkey(r.poDate);om[k]=(om[k]||0)+r.orderQty;seen[k]=1}
      if(r.dispDate&&r.sentQty){var d=mkey(r.dispDate);dm[d]=(dm[d]||0)+r.sentQty;dc[d]=(dc[d]||0)+1;seen[d]=1}
    });
    var ml=Object.keys(seen).sort();
    if(!ml.length)empty("cvMonth","No dated lines for these filters.");
    else draw("cvMonth",{type:"bar",data:{labels:ml.map(mlabel),datasets:[
      {label:"Order qty (PO month)",data:ml.map(function(k){return om[k]||0}),backgroundColor:ml.map(function(k){return flt.month==="all"||flt.month===k?acc:acc2}),borderRadius:4},
      {label:"Dispatched qty (dispatch month)",data:ml.map(function(k){return dm[k]||0}),backgroundColor:good,borderRadius:4}]},
      options:{plugins:{legend:{position:"bottom"},tooltip:{callbacks:{label:function(c){return " "+c.dataset.label+": "+fmt(c.parsed.y)},
        afterLabel:function(c){return c.datasetIndex===1?" "+(dc[ml[c.dataIndex]]||0)+" dispatch lines":""}}}},
        scales:{y:{ticks:{callback:function(v){return fmt(v)}},grid:{color:line}},x:{grid:{display:false}}}}});

    var pc={}; rows.forEach(function(r){if(r.pending&&r.pendQty)pc[r.company]=(pc[r.company]||0)+r.pendQty});
    var pl=Object.keys(pc).sort(function(a,b){return pc[b]-pc[a]}).slice(0,10);
    if(!pl.length)empty("cvPending","Nothing pending for these filters.");
    else draw("cvPending",{type:"bar",data:{labels:pl,datasets:[{label:"Pending qty",data:pl.map(function(c){return pc[c]}),backgroundColor:warn,borderRadius:4}]},
      options:{indexAxis:"y",plugins:{legend:{display:false},tooltip:{callbacks:{label:function(c){return " "+fmt(c.parsed.x)}}}},
        scales:{x:{ticks:{callback:function(v){return fmt(v)}},grid:{color:line}},y:{ticks:{callback:function(v){return trunc(this.getLabelForValue(v),24)}},grid:{display:false}}}}});

    var pr=rows.filter(function(r){return r.pending&&r.pendQty}).sort(function(a,b){return b.pendQty-a.pendQty});
    $("pendHint").textContent=pr.length?("Showing "+Math.min(pr.length,100)+" of "+fmt(pr.length)+" pending lines, largest first."):"No pending lines for these filters.";
    $("pendBody").innerHTML=pr.slice(0,100).map(function(r){
      return "<tr><td>"+esc(r.company)+"</td><td>"+esc(r.po)+"</td><td>"+esc(r.poDateRaw)+"</td><td>"+esc(r.pendItem||"-")+'</td><td class="num"><span class="pill">'+fmt(r.pendQty)+"</span></td></tr>";
    }).join("");
  }

  load();
})();
